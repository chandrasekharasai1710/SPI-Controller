`timescale 1ns/1ps

module tb_spi_master;

    reg clk;
    reg rst_n;
    reg start;
    reg [7:0] tx_data;

    wire [7:0] rx_data;
    wire done;
    wire sclk;
    wire mosi;
    wire cs_n;

    reg [7:0] slave_tx_shift;
    reg [7:0] slave_rx_shift;
    reg [7:0] slave_tx_byte;
    reg miso;

    integer errors;
    integer tests;

    assign miso = slave_tx_shift[7];

    spi_master #(
        .CLK_DIV(4)
    ) dut (
        .clk(clk),
        .rst_n(rst_n),
        .start(start),
        .tx_data(tx_data),
        .rx_data(rx_data),
        .done(done),
        .sclk(sclk),
        .mosi(mosi),
        .miso(miso),
        .cs_n(cs_n)
    );

    always #5 clk = ~clk;

    always @(posedge sclk) begin
        if (!cs_n)
            slave_rx_shift <= {slave_rx_shift[6:0], mosi};
    end

    always @(negedge sclk) begin
        if (!cs_n)
            slave_tx_shift <= {slave_tx_shift[6:0], 1'b0};
    end

    task run_transfer;
        input [7:0] master_byte;
        input [7:0] slave_byte;
        begin
            @(negedge clk);
            tx_data = master_byte;
            slave_tx_byte = slave_byte;
            slave_tx_shift = slave_byte;
            slave_rx_shift = 8'b0;
            start = 1'b1;

            @(negedge clk);
            start = 1'b0;

            fork
                begin
                    wait (done === 1'b1);
                end
                begin
                    repeat (1000) @(posedge clk);
                    $fatal(1, "TIMEOUT: SPI transfer did not complete");
                end
            join_any
            disable fork;

            tests = tests + 1;

            if (rx_data !== slave_tx_byte) begin
                $display("FAIL RX: TX=%02h EXPECTED=%02h GOT=%02h",
                         master_byte, slave_tx_byte, rx_data);
                errors = errors + 1;
            end else begin
                $display("PASS RX: TX=%02h RX=%02h",
                         master_byte, rx_data);
            end

            if (slave_rx_shift !== master_byte) begin
                $display("FAIL MOSI: EXPECTED=%02h GOT=%02h",
                         master_byte, slave_rx_shift);
                errors = errors + 1;
            end else begin
                $display("PASS MOSI: RECEIVED=%02h", slave_rx_shift);
            end

            @(negedge clk);
        end
    endtask

    initial begin
        clk = 1'b0;
        rst_n = 1'b0;
        start = 1'b0;
        tx_data = 8'b0;
        slave_tx_shift = 8'b0;
        slave_rx_shift = 8'b0;
        slave_tx_byte = 8'b0;
        errors = 0;
        tests = 0;

        repeat (4) @(negedge clk);
        rst_n = 1'b1;

        repeat (2) @(negedge clk);

        run_transfer(8'hA5, 8'h3C);
        run_transfer(8'h00, 8'hFF);
        run_transfer(8'hFF, 8'h00);
        run_transfer(8'h55, 8'hAA);
        run_transfer(8'h81, 8'h7E);

        $display("--------------------------------------");
        $display("TRANSFERS: %0d", tests);
        $display("ERRORS: %0d", errors);

        if (errors == 0)
            $display("RESULT: ALL TESTS PASSED");
        else
            $fatal(1, "RESULT: TESTS FAILED");

        $finish;
    end

    initial begin
        $dumpfile("spi_master.vcd");
        $dumpvars(0, tb_spi_master);
    end

endmodule
